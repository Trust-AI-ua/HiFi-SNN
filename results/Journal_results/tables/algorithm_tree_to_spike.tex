% ======================================================================
% Training-objective equations (reference the algorithm)
% ======================================================================
The student is trained with a five-term objective. With
$\sigma(\mathbf{z})$ the softmax and
$\tilde{\mathbf{p}} \propto \mathbf{p}^{1/\tau}$ the temperature-sharpened
teacher distribution,
\begin{align}
\mathcal{L}_{\mathrm{CE}}   &= -\log \sigma(\mathbf{z})_{y}, \\
\mathcal{L}_{\mathrm{KD}}   &= -\sum_{k=0}^{K-1} \tilde{p}_{k}
                               \log \sigma(\mathbf{z})_{k}, \\
\mathcal{L}_{\mathrm{rank}} &= \frac{1}{B}\sum_{i=1}^{B}
      \max\!\Bigl(0,\; \mu - \operatorname{sign}\!\bigl(a^{\mathcal{G}}_{i}
      - a^{\mathcal{G}}_{\pi(i)}\bigr)\bigl(a^{f}_{i} - a^{f}_{\pi(i)}\bigr)\Bigr),
      \label{eq:rank}\\
\mathcal{L}_{\mathrm{sp}}   &= \rho, \\
\mathcal{L}_{\mathrm{FP}}   &= \frac{1}{|\mathcal{B}|}\sum_{i \in \mathcal{B}}
      \bigl(1 - \sigma(\mathbf{z}_{i})_{0}\bigr),
\end{align}
where $a^{f} = 1 - \sigma(\mathbf{z})_{0}$ and
$a^{\mathcal{G}} = 1 - p_{0}$ are the student and teacher attack scores and
$\mathcal{B}$ is the set of benign samples in the batch. The ranking term
(Eq.~\ref{eq:rank}) preserves the teacher's ordering of attack likelihood; the
sparsity term penalises hidden firing to reduce synaptic operations; the FP
term keeps the student confident on true-benign flows to limit alert fatigue.
The total loss is
\begin{equation}
\mathcal{L} = \mathcal{L}_{\mathrm{CE}}
  + \alpha_{\mathrm{kd}}   \mathcal{L}_{\mathrm{KD}}
  + \alpha_{\mathrm{rank}} \mathcal{L}_{\mathrm{rank}}
  + \alpha_{\mathrm{sp}}   \mathcal{L}_{\mathrm{sp}}
  + \alpha_{\mathrm{fp}}   \mathcal{L}_{\mathrm{FP}}.
\label{eq:total_loss}
\end{equation}

% ======================================================================
% Algorithm 1: training
% ======================================================================
\begin{algorithm}[!t]
\caption{Tree-to-Spike training (teacher, encoder, distillation)}
\label{alg:treespike_train}
\begin{algorithmic}[1]
\Require training set $\{(\mathbf{x}_i, y_i)\}_{i=1}^{N}$, window $T$,
         budget $M_{\max}$, prune quantile $q$, jitter $\delta$,
         loss weights $(\alpha_{\mathrm{kd}}, \alpha_{\mathrm{rank}},
         \alpha_{\mathrm{sp}}, \alpha_{\mathrm{fp}})$, temperature $\tau$
\Ensure fixed encoder $\Phi$, int8-quantised student $f_{\theta}$
\Statex \textbf{Phase 1: cost-sensitive teacher (training only)}
\State $w_i \gets \bigl(N / n_{y_i}\bigr)^{\gamma}$ for all $i$
       \Comment{up-weight rare classes}
\State fit XGBoost teacher $\mathcal{G}$ on
       $\{(\mathbf{x}_i, y_i, w_i)\}$
\Statex \textbf{Phase 2: tree-guided encoder construction}
\State $\mathcal{E}_j \gets$ sorted unique split thresholds of feature $j$
       over all trees of $\mathcal{G}$, for $j = 1,\dots,F$
\State $g \gets$ gain importances of $\mathcal{G}$;\;
       $\theta_{\mathrm{keep}} \gets q\text{-quantile of } \{g_j : g_j > 0\}$
\For{$j = 1,\dots,F$}
  \If{$g_j \geq \theta_{\mathrm{keep}}$ \textbf{and} $\mathcal{E}_j \neq \emptyset$}
    \State $m_j \gets \max\!\bigl(1, \lceil (g_j / g_{\max}) M_{\max} \rceil\bigr)$
    \State add $m_j$ channels for feature $j$ with offsets
           $o_c = \bigl(c - \tfrac{m_j-1}{2}\bigr)\delta$
  \EndIf \Comment{features below $\theta_{\mathrm{keep}}$ are pruned}
\EndFor
\State $C \gets$ total channels; freeze $\Phi$
       \Comment{the tree is never executed again}
\Statex \textbf{Phase 3: spike encoding} (definition of $\Phi$)
\State $\ell_j(\mathbf{x}) \gets \mathrm{rank}(x_j;\mathcal{E}_j)/|\mathcal{E}_j|$;\;
       $\ell_c \gets \mathrm{clip}(\ell_j + o_c,\, 0,\, 1)$
\State $t_c \gets \mathrm{round}\bigl((1-\ell_c)(T-1)\bigr)$;\;
       $S_{t,c} \gets \mathbf{1}[t = t_c]$
       \Comment{TTFS: informative values fire early}
\Statex \textbf{Phase 4: distillation of the SNN student}
\For{each epoch, each mini-batch $(\mathbf{X}, \mathbf{y})$}
  \State $\mathbf{S} \gets \Phi(\mathbf{X})$;\;
         $\mathbf{p} \gets \mathcal{G}(\mathbf{X})$
         \Comment{spikes and teacher soft labels}
  \State $(\mathbf{z}, \rho) \gets f_{\theta}(\mathbf{S})$
         \Comment{LIF forward over $T$ steps, surrogate gradients}
  \State update $\theta$ by descending
         $\nabla_{\theta}\mathcal{L}$ of Eq.~\ref{eq:total_loss}
\EndFor
\Statex \textbf{Phase 5: deployment preparation}
\State $W \gets \mathrm{round}(W / s) \cdot s$ with $s = \max|W|/127$
       for every linear layer \Comment{int8 quantisation}
\State \Return $(\Phi, f_{\theta})$; discard $\mathcal{G}$
\end{algorithmic}
\end{algorithm}

% ======================================================================
% Algorithm 2: deployment inference with early exit
% ======================================================================
\begin{algorithm}[!t]
\caption{Deployment inference with confidence-triggered early exit
         (no tree at runtime)}
\label{alg:treespike_deploy}
\begin{algorithmic}[1]
\Require flow $\mathbf{x}$, fixed encoder $\Phi$, quantised student
         $f_{\theta}$, thresholds $\theta_{\mathrm{exit}} > \theta_{\mathrm{unc}}$
\Ensure prediction $\hat{y}$, exit step $t^{*}$, uncertain flag $u$
\State $\mathbf{S} \gets \Phi(\mathbf{x})$
       \Comment{fixed threshold lookup, $O(F \log |\mathcal{E}_j|)$}
\State $\bar{U} \gets \mathbf{0}$
\For{$t = 1,\dots,T$}
  \State propagate $\mathbf{S}_{t}$ through the LIF layers;
         $\bar{U} \gets \bar{U} + U^{(t)}_{3}$
  \State $\mathbf{q} \gets \sigma\bigl(\bar{U}/t\bigr)$
         \Comment{running class posterior}
  \If{$\max_k q_k \geq \theta_{\mathrm{exit}}$}
    \State \Return $\hat{y} = \arg\max_k q_k$,\; $t^{*} = t$,\; $u = 0$
           \Comment{early exit: skip remaining steps}
  \EndIf
\EndFor
\State $\mathbf{q} \gets \sigma\bigl(\bar{U}/T\bigr)$;\;
       $\hat{y} \gets \arg\max_k q_k$;\; $t^{*} \gets T$
\State $u \gets \mathbf{1}\bigl[\max_k q_k < \theta_{\mathrm{unc}}\bigr]$
       \Comment{flag low-confidence verdicts for analyst review}
\State \Return $\hat{y}$, $t^{*}$, $u$
\end{algorithmic}
\end{algorithm}
